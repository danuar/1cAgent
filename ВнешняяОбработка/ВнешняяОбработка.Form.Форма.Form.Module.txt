// =========================================================
// Форма-раннер 1С-агента.
//
// Обмен с питон-стороной идёт ЧЕРЕЗ ФАЙЛЫ в "папке обмена":
//   error.txt      — состояние задания: "WAIT" (питон ждёт) -> "OK"/"START"/текст ошибки
//   AgentCode.epf  — собранная питоном обработка с процедурой ВыполнитьЗадачу
//   logs.txt       — накопительный лог (питон читает ДЕЛЬТУ после своего задания)
//
// #64, папка обмена. Раньше она бралась ТОЛЬКО из ПараметрЗапуска (ключ /C), и
// при открытии через Файл->Открыть форма молча выходила — выглядело как
// "обработка не работает". Теперь порядок такой, первое непустое побеждает:
//   1) ПараметрЗапуска — /C из батника и start_runner (как было);
//   2) настройка, сохранённая В ЭТОЙ БАЗЕ (ХранилищеОбщихНастроек) — своя у
//      каждого пользователя и каждой базы;
//   3) то, что человек ввёл в поле руками.
//
// #64, ВАЖНО ПРО УСТРОЙСТВО: реквизиты, поля, команды и кнопки формы
// создаются ПРОГРАММНО в ПодготовитьФорму() — в Form.xml не добавлено ничего.
// Причина сугубо практическая: собрать .epf из иерархической выгрузки через
// конфигуратор в пакетном режиме не выходит (LoadExternalDataProcessorOrReport
// FromFiles берёт только ФАЙЛ и считает его ПЛОСКОЙ выгрузкой, каталог не
// принимает), поэтому единственное, что нужно перенести в .epf руками, — этот
// модуль. Следствие: к динамическим реквизитам обращаемся ТОЛЬКО через
// ЭтотОбъект["Имя"] — по имени напрямую компилятор их не видит.
// =========================================================

&НаКлиенте
Процедура ЗаписатьВЛог(ТекстЛога, ПутьКФайлу)
    ФайлЛога = Новый ТекстовыйДокумент;

    // Проверяем, существует ли файл. Если да - читаем его, чтобы дописать в конец
    ФайлПроверка = Новый Файл(ПутьКФайлу);
    Если ФайлПроверка.Существует() Тогда
        ФайлЛога.Прочитать(ПутьКФайлу, КодировкаТекста.UTF8);
    КонецЕсли;

    ФайлЛога.ДобавитьСтроку(Строка(ТекущаяДата()) + ": " + ТекстЛога);
    ФайлЛога.Записать(ПутьКФайлу, КодировкаТекста.UTF8);
КонецПроцедуры

&НаКлиенте
Процедура ЗапуститьАгента(Команда)
    КорневаяПапка = СокрЛП(ЭтотОбъект["ПапкаОбмена"]);
    Если КорневаяПапка = "" Тогда
        УстановитьСостояние("Папка обмена не задана");
        Возврат;
    КонецЕсли;

    // #71: адрес вида http://хост:порт#токен — работаем по HTTP, раннер сам
    // ходит к агенту (см. ОпроситьАгента). Иначе это путь к папке обмена.
    НижнийРегистр = НРег(КорневаяПапка);
    Если Лев(НижнийРегистр, 7) = "http://" Или Лев(НижнийРегистр, 8) = "https://" Тогда
        ОпроситьАгента(КорневаяПапка);
        Возврат;
    КонецЕсли;

    ПутьКЛогуОшибок = КорневаяПапка + "\error.txt";
    ПутьКИстории = КорневаяПапка + "\logs.txt";
    ПутьКEPF = КорневаяПапка + "\AgentCode.epf";

    // Проверка статуса от Питона
    ФайлСтатуса = Новый Файл(ПутьКЛогуОшибок);
    Если Не ФайлСтатуса.Существует() Тогда Возврат; КонецЕсли;

    ДокСтатус = Новый ТекстовыйДокумент;
    ДокСтатус.Прочитать(ПутьКЛогуОшибок, КодировкаТекста.UTF8);
    ТекущийСтатус = СокрЛП(ДокСтатус.ПолучитьТекст());

    // #67: ДВА протокола заданий.
    //   "WAIT"      — как было: питон собрал AgentCode.epf, подключаем его.
    //   "WAIT_EXEC" — БЫСТРЫЙ ПУТЬ: код пришёл ТЕКСТОМ в task.bsl, конфигуратор
    //                 на стороне питона НЕ запускался (это ~3.4с из ~5с всего
    //                 цикла). Годится, когда в задании нет объявлений своих
    //                 процедур/функций — решает питон, см. glue._direct_body.
    // Старый раннер реагирует ТОЛЬКО на "WAIT", поэтому "WAIT_EXEC" он просто
    // не заметит, а питон по таймауту откатится на сборку .epf.
    Если ТекущийСтатус = "WAIT_EXEC" Тогда
        ПутьКЗаданию = КорневаяПапка + "\task.bsl";
        ФайлЗадания = Новый Файл(ПутьКЗаданию);
        Если Не ФайлЗадания.Существует() Тогда Возврат; КонецЕсли;
        ДокЗадание = Новый ТекстовыйДокумент;
        ДокЗадание.Прочитать(ПутьКЗаданию, КодировкаТекста.UTF8);
        Результат = ВыполнитьТекстНаСервере(ДокЗадание.ПолучитьТекст(), ЭтотОбъект["БезопасныйРежим"]);

    ИначеЕсли ТекущийСтатус = "WAIT" Тогда
        ФайлОбработки = Новый Файл(ПутьКEPF);
        Если Не ФайлОбработки.Существует() Тогда Возврат; КонецЕсли;

        // Читаем EPF в бинарном виде на Клиенте и кладем в хранилище
        ДвоичныеДанныеEPF = Новый ДвоичныеДанные(ПутьКEPF);
        АдресВХранилище = ПоместитьВоВременноеХранилище(ДвоичныеДанныеEPF, УникальныйИдентификатор);
        Результат = ВыполнитьОбработкуНаСервере(АдресВХранилище, ЭтотОбъект["БезопасныйРежим"]);

    Иначе
        Возврат;   // питон ничего не просил
    КонецЕсли;

    Если Не ПустаяСтрока(Результат.ЛогИИ) Тогда
        ЗаписатьВЛог(Результат.ЛогИИ, ПутьКИстории);
    КонецЕсли;

    ОтветПитону = Новый ТекстовыйДокумент;
    Если ПустаяСтрока(Результат.ОшибкаСистемы) Тогда
        Если Найти(Результат.ЛогИИ, "ГОТОВО") > 0 Тогда
            ОтветПитону.ДобавитьСтроку("OK");
        Иначе
            ОтветПитону.ДобавитьСтроку("START");
        КонецЕсли;
    Иначе
        ОтветПитону.ДобавитьСтроку(Результат.ОшибкаСистемы);
        ЗаписатьВЛог("ОШИБКА: " + Результат.ОшибкаСистемы, ПутьКИстории);
    КонецЕсли;

    ОтветПитону.Записать(ПутьКЛогуОшибок, КодировкаТекста.UTF8);
    УстановитьСостояние("Последнее задание: " + Строка(ТекущаяДата()));
КонецПроцедуры

&НаСервереБезКонтекста
Функция ВыполнитьОбработкуНаСервере(ПутьКФайлу, БезопасныйРежим)
    ЛогВыполнения = "";
    ТекстОшибки = "";

    Попытка
        // #64: третьим параметром идёт БЕЗОПАСНЫЙ РЕЖИМ. Раньше он не
        // передавался вовсе, то есть был Истина по умолчанию — а в нём
        // ЗАПРЕЩЕНО непосредственное удаление объектов ("Действие недоступно
        // в безопасном режиме"), при том что чтение и запись работают.
        // Четвёртый параметр глушит модальное предупреждение об опасных
        // действиях, иначе снятие безопасного режима вешает диалог.
        Защита = Новый ОписаниеЗащитыОтОпасныхДействий;
        Защита.ПредупреждатьОбОпасныхДействиях = Ложь;
        ИмяОбработки = ВнешниеОбработки.Подключить(ПутьКФайлу, , БезопасныйРежим, Защита);
        ОбработкаОбъект = ВнешниеОбработки.Создать(ИмяОбработки);

        // Вызываем главную процедуру ИИ
        ОбработкаОбъект.ВыполнитьЗадачу(ЛогВыполнения);

    Исключение
        ТекстОшибки = ПодробноеПредставлениеОшибки(ИнформацияОбОшибке());
    КонецПопытки;

    Для Каждого Сообщение Из ПолучитьСообщенияПользователю(Истина) Цикл
        ЛогВыполнения = ЛогВыполнения + Символы.ПС + Сообщение.Текст;
    КонецЦикла;

    Результат = Новый Структура("ЛогИИ, ОшибкаСистемы", ЛогВыполнения, ТекстОшибки);
    Возврат Результат;
КонецФункции

&НаКлиенте
Функция ВыполнитьТекстНаКлиенте(ТекстКода)
    // #90: то же, что ВыполнитьТекстНаСервере, но код исполняется В КЛИЕНТСКОМ
    // контексте сеанса раннера. Нужно ТОЛЬКО для заданий, работающих с формами:
    // в клиент-серверной базе серверный контекст про ОткрытьФорму/ПолучитьФорму
    // не знает вовсе (в файловой базе это не мешает — там один процесс).
    //
    // Ограничения платформы те же, что у быстрого пути: Выполнить() исполняет код
    // в контексте ЭТОЙ функции, поэтому внутри нельзя объявлять процедуры/функции
    // и нельзя Возврат. Питон проверяет это заранее (glue._direct_body).
    //
    // Безопасный режим здесь НЕ включается: УстановитьБезопасныйРежим — серверный
    // метод. На клиенте код и так ограничен правами пользователя сеанса.
    ЛогВыполнения = "";
    ТекстОшибки = "";

    Попытка
        Выполнить(ТекстКода);
    Исключение
        ТекстОшибки = ПодробноеПредставлениеОшибки(ИнформацияОбОшибке());
    КонецПопытки;

    Для Каждого Сообщение Из ПолучитьСообщенияПользователю(Истина) Цикл
        ЛогВыполнения = ЛогВыполнения + Символы.ПС + Сообщение.Текст;
    КонецЦикла;

    Возврат Новый Структура("ЛогИИ, ОшибкаСистемы", ЛогВыполнения, ТекстОшибки);
КонецФункции

// =========================================================
// #71: HTTP-КЛИЕНТ РАННЕРА. Альтернатива папке обмена — раннер сам ходит к
// агенту, поэтому может жить на чужом сервере за NAT, куда не протянуть ни
// файловую шару, ни VPN.
//
// В поле "Папка обмена" вместо пути пишется адрес вида
//     http://хост:порт#токен
// Токен агент кладёт в work/http_token.txt при http_transport_start().
//
// ПРО ИНТЕРВАЛ ОПРОСА. 1С не гарантирует переиспользование соединения —
// каждый опрос это НОВОЕ рукопожатие (через интернет с TLS это 100-300мс).
// Поэтому интервал адаптивный: 1с пока идут задания, +1с на каждый пустой
// ответ, потолок 5с. Долбить сеть раз в секунду круглосуточно смысла нет.
// Длинный опрос (wait=2) специально короткий: обработчик ожидания в 1С
// СИНХРОННЫЙ и блокирует сессию на время запроса.
// =========================================================

&НаКлиенте
Функция РазобратьАдресАгента(Знач Адрес)
    Результат = Новый Структура("Схема, Хост, Порт, Токен", "http", "", 80, "");
    Текст = СокрЛП(Адрес);

    Поз = Найти(Текст, "#");
    Если Поз > 0 Тогда
        Результат.Токен = СокрЛП(Сред(Текст, Поз + 1));
        Текст = Лев(Текст, Поз - 1);
    КонецЕсли;

    Если Лев(НРег(Текст), 8) = "https://" Тогда
        Результат.Схема = "https";
        Результат.Порт = 443;
        Текст = Сред(Текст, 9);
    ИначеЕсли Лев(НРег(Текст), 7) = "http://" Тогда
        Текст = Сред(Текст, 8);
    КонецЕсли;

    Поз = Найти(Текст, "/");
    Если Поз > 0 Тогда
        Текст = Лев(Текст, Поз - 1);
    КонецЕсли;

    Поз = Найти(Текст, ":");
    Если Поз > 0 Тогда
        Результат.Хост = Лев(Текст, Поз - 1);
        Результат.Порт = Число(Сред(Текст, Поз + 1));
    Иначе
        Результат.Хост = Текст;
    КонецЕсли;

    Возврат Результат;
КонецФункции

&НаКлиенте
Функция СоединениеСАгентом(Разбор)
    Защита = Неопределено;
    Если Разбор.Схема = "https" Тогда
        Защита = Новый ЗащищенноеСоединениеOpenSSL;
    КонецЕсли;
    // Таймаут заведомо больше wait=2 длинного опроса, иначе будем рвать
    // собственный запрос раньше, чем агент успеет ответить.
    Возврат Новый HTTPСоединение(Разбор.Хост, Разбор.Порт, , , , 20, Защита);

    Возврат Значение;

&НаКлиенте
Процедура ЗамедлитьОпрос()
    // #72 (по замеру): замедлять ТИКИ — плохая идея. Пока раннер спит между
    // опросами, задание лежит на доске и ждёт его пробуждения: так  по
    // HTTP отставал от файлов на 1.2с. Правильнее держать раннера ВНУТРИ
    // длинного опроса — тогда задание уходит ему мгновенно (замерено: 0.02с
    // против 1.01с), а рукопожатий даже меньше, чем при откате тиков до 5с.
    // Поэтому в простое растёт ДЛИТЕЛЬНОСТЬ ОЖИДАНИЯ запроса, а не пауза.
    Если Не ЭтотОбъект["БыстрыйОпрос"] Тогда
        Возврат;   // в отзывчивом режиме запрос и так мгновенный, растить нечего
    КонецЕсли;
    Следующее = ЭтотОбъект["ОжиданиеЗапроса"] + 1;
    Если Следующее > 5 Тогда
        Следующее = 5;
    КонецЕсли;
    ЭтотОбъект["ОжиданиеЗапроса"] = Следующее;

Процедура ОтправитьРезультатАгенту(Разбор, НомерЗадания, Результат)
    Статус = "OK";
    Если Не ПустаяСтрока(Результат.ОшибкаСистемы) Тогда
        Статус = Результат.ОшибкаСистемы;
    ИначеЕсли Найти(Результат.ЛогИИ, "ГОТОВО") = 0 Тогда
        Статус = "START";
    КонецЕсли;

    Тело = Новый Структура("seq, status, log", НомерЗадания, Статус, Результат.ЛогИИ);

    ЗаписатьJSON(Запись, Тело);
    ТелоСтрокой = Запись.Закрыть();

    // старый объект смысла нет, а вот получить на нём разорванный сокет — легко.
    Отправка = Новый HTTPЗапрос("/result");
    Отправка.Заголовки.Вставить("X-Agent-Token", Разбор.Токен);
    Отправка.Заголовки.Вставить("Content-Type", "application/json; charset=utf-8");
    Отправка.УстановитьТелоИзСтроки(ТелоСтрокой, КодировкаТекста.UTF8);
    СоединениеСАгентом(Разбор).ОтправитьДляОбработки(Отправка);

&НаКлиенте
Процедура ОпроситьАгента(Знач Адрес)
    // #72: после выполненного задания НЕ уходим спать до следующего тика —
    // сразу спрашиваем следующее. Задания от агента обычно идут пачкой, и
    // секунда ожидания тика между ними была чистой потерей (замер: exec по
    // HTTP стабильно 1.00с, из которых работа — миллисекунды). Цикл ограничен
    // сверху, чтобы сессия не зависла навсегда в обработчике ожидания.
    Для Попытка_ = 1 По 5 Цикл
        Если Не ОпроситьАгентаОднократно(Адрес) Тогда
            Прервать;   // заданий нет, агент недоступен или ошибка — ждём тик
        КонецЕсли;
    КонецЦикла;
КонецПроцедуры

&НаКлиенте
Функция ОпроситьАгентаОднократно(Знач Адрес)
    Разбор = РазобратьАдресАгента(Адрес);
    Если ПустаяСтрока(Разбор.Хост) Или ПустаяСтрока(Разбор.Токен) Тогда
        УстановитьСостояние("Адрес агента должен быть вида http://хост:порт#токен");
        Возврат Ложь;
    КонецЕсли;

    Попытка
        // #74: КЛЮЧЕВОЙ момент для отзывчивости сеанса. Обработчик ожидания в
        // 1С СИНХРОННЫЙ — он блокирует сеанс на всё время HTTP-запроса. При
        // длинном опросе (wait 2-5с) раннер получает задание мгновенно, но
        // человек в этом же сеансе работать не может: окно подвисает почти
        // постоянно. Поэтому по умолчанию wait=0 — сервер отвечает сразу,
        // блокировка равна одному круговому времени запроса. Длинный опрос
        // включается флажком и только для ВЫДЕЛЕННОГО сеанса раннера.
        ДлительностьОжидания = 0;
        Если ЭтотОбъект["БыстрыйОпрос"] Тогда
            ДлительностьОжидания = ЭтотОбъект["ОжиданиеЗапроса"];
        КонецЕсли;
        Запрос = Новый HTTPЗапрос("/task?wait=" + Формат(ДлительностьОжидания, "ЧН=0; ЧГ=0"));
        Запрос.Заголовки.Вставить("X-Agent-Token", Разбор.Токен);
        Ответ = СоединениеСАгентом(Разбор).Получить(Запрос);
    Исключение
        // Агента сейчас нет — это ШТАТНОЕ состояние, а не ошибка: он живёт
        // внутри процесса Claude и поднимается только на время работы.
        УстановитьСостояние("Агент недоступен, жду");
        ЗамедлитьОпрос();
        Возврат Ложь;
    КонецПопытки;

    Если Ответ.КодСостояния = 204 Тогда
        УстановитьСостояние("Заданий нет");
        ЗамедлитьОпрос();
        Возврат Ложь;
    ИначеЕсли Ответ.КодСостояния = 502 Или Ответ.КодСостояния = 503 Или Ответ.КодСостояния = 504 Тогда
        // #80: прокси на месте, а агента за ним нет — это ШТАТНОЕ состояние
        // (агент живёт внутри процесса Claude). Раньше показывался голый код
        // 502, и выглядело как поломка.
        УстановитьСостояние("Агент не запущен, жду");
        ЗамедлитьОпрос();
        Возврат Ложь;
    ИначеЕсли Ответ.КодСостояния = 401 Тогда
        УстановитьСостояние("Неверный токен — проверьте адрес после #");
        ЗамедлитьОпрос();
        Возврат Ложь;
    ИначеЕсли Ответ.КодСостояния <> 200 Тогда
        УстановитьСостояние("Агент ответил кодом " + Ответ.КодСостояния);
        ЗамедлитьОпрос();
        Возврат Ложь;
    КонецЕсли;

    Задание = РазобратьJSON(Ответ.ПолучитьТелоКакСтроку("UTF-8"));
    НомерЗадания = Задание.Получить("seq");

    Если Задание.Получить("mode") = "exec" Тогда
        // #90: агент может попросить КЛИЕНТСКИЙ контекст — это единственный способ
        // работать с формами (ОткрытьФорму/ПолучитьФорму) в клиент-серверной базе.
        // Старый агент поля "client" не шлёт, Получить() вернёт Неопределено -> сервер.
        Если Задание.Получить("client") = Истина Тогда
            Результат = ВыполнитьТекстНаКлиенте(Задание.Получить("code"));
        Иначе
            Результат = ВыполнитьТекстНаСервере(Задание.Получить("code"), ЭтотОбъект["БезопасныйРежим"]);
        КонецЕсли;
    Иначе
        ДвоичныеДанные = Base64Значение(Задание.Получить("epf_b64"));
        АдресВХранилище = ПоместитьВоВременноеХранилище(ДвоичныеДанные, УникальныйИдентификатор);
        Результат = ВыполнитьОбработкуНаСервере(АдресВХранилище, ЭтотОбъект["БезопасныйРежим"]);
    КонецЕсли;

    Попытка
        ОтправитьРезультатАгенту(Разбор, НомерЗадания, Результат);
        УстановитьСостояние("Задание выполнено: " + Строка(ТекущаяДата()));
    Исключение
        УстановитьСостояние("Результат отправить не удалось: "
            + КраткоеПредставлениеОшибки(ИнформацияОбОшибке()));
    КонецПопытки;

    ЭтотОбъект["ОжиданиеЗапроса"] = 2;   // задания пошли — короткие запросы
    Возврат Истина;
КонецФункции

&НаКлиенте
Процедура ПриОткрытии(Отказ)
    ПодготовитьФорму();

    Настройки = ПрочитатьНастройкиНаСервере();
    ЭтотОбъект["БезопасныйРежим"] = Настройки.БезопасныйРежим;
    ЭтотОбъект["БыстрыйОпрос"] = Настройки.БыстрыйОпрос;

    // 1) /C при запуске, 2) настройка этой базы, 3) руками в поле
    Если Не ПустаяСтрока(СокрЛП(ПараметрЗапуска)) Тогда
        ЭтотОбъект["ПапкаОбмена"] = СокрЛП(ПараметрЗапуска);
        // #76: раз раннера поднял сам агент ключом /C — это ВЫДЕЛЕННЫЙ сеанс,
        // человек в нём не работает, подвисания никому не мешают. Включаем
        // длинный опрос: задание доставляется мгновенно (0.01с против ~1с).
        ЭтотОбъект["БыстрыйОпрос"] = Истина;
    Иначе
        ЭтотОбъект["ПапкаОбмена"] = Настройки.Папка;
    КонецЕсли;

    Если ПустаяСтрока(СокрЛП(ЭтотОбъект["ПапкаОбмена"])) Тогда
        УстановитьСостояние("Папка обмена не задана — укажите её и нажмите «Сохранить настройки»");
        Возврат;
    КонецЕсли;

    // #67: сообщаем питону, что этот раннер понимает быстрый путь. Файла нет —
    // значит раннер старый, и питон не будет пробовать "WAIT_EXEC" вслепую.
    Попытка
        Объявление = Новый ТекстовыйДокумент;
        Объявление.ДобавитьСтроку("protocol=2");
        Объявление.Записать(СокрЛП(ЭтотОбъект["ПапкаОбмена"]) + "\runner.txt", КодировкаТекста.UTF8);
    Исключение
        // не смогли — не страшно, просто останемся на медленном пути
    КонецПопытки;

	ВключитьЦикличныйПерезапуск(Неопределено);
КонецПроцедуры
